import 'dart:async';

import 'package:flutter/material.dart';
import 'package:just_audio/just_audio.dart';

import 'models/song.dart';
import 'services/music_service.dart';

void main() {
  runApp(const SoundWaveApp());
}

const Color backgroundColor = Color(0xFF121212);
const Color cardColor = Color(0xFF242424);
const Color accentColor = Color(0xFFB8F34A);

class SoundWaveApp extends StatelessWidget {
  const SoundWaveApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'SoundWave',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        brightness: Brightness.dark,
        scaffoldBackgroundColor: backgroundColor,
        colorScheme: ColorScheme.fromSeed(
          seedColor: accentColor,
          brightness: Brightness.dark,
        ),
        useMaterial3: true,
      ),
      home: const HomePage(),
    );
  }
}


class HomePage extends StatefulWidget {
  const HomePage({super.key});

  @override
  State<HomePage> createState() => _HomePageState();
}

class _HomePageState extends State<HomePage> {
  final TextEditingController searchController = TextEditingController();
  final AudioPlayer audioPlayer = AudioPlayer();
  final MusicService musicService = MusicService();

  List<Song> homeSongs = [];
  List<Song> searchResults = [];
  final Map<int, Song> favoriteSongs = {};

  bool isHomeLoading = true;
  bool isSearchLoading = false;
  String? homeError;
  String? searchError;

  int currentIndex = 0;
  int? playingSongId;
  Song? currentSong;
  String lastSearchTerm = '';

  @override
  void initState() {
    super.initState();
    fetchHomeSongs();
  }

  Future<List<Song>> requestSongs(String term) {
    return musicService.requestSongs(term);
  }

  Future<void> fetchHomeSongs() async {
    setState(() {
      isHomeLoading = true;
      homeError = null;
    });

    try {
      final loadedSongs = await requestSongs('pop music');
      if (!mounted) return;
      setState(() {
        homeSongs = loadedSongs;
        isHomeLoading = false;
        if (homeSongs.isEmpty) {
          homeError = 'Lagu belum ditemukan. Coba muat ulang.';
        }
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        isHomeLoading = false;
        homeError = 'Gagal mengambil lagu. Periksa koneksi internet.';
      });
    }
  }

  Future<void> searchSongs() async {
    final term = searchController.text.trim();
    if (term.isEmpty) {
      setState(() {
        searchResults = [];
        searchError = 'Ketik judul lagu atau nama artis terlebih dahulu.';
      });
      return;
    }

    FocusScope.of(context).unfocus();
    setState(() {
      lastSearchTerm = term;
      isSearchLoading = true;
      searchError = null;
    });

    try {
      final results = await requestSongs(term);
      if (!mounted) return;
      setState(() {
        searchResults = results;
        isSearchLoading = false;
        if (results.isEmpty) {
          searchError = 'Tidak ada lagu yang cocok dengan "$term".';
        }
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        isSearchLoading = false;
        searchError = 'Pencarian gagal. Periksa koneksi internet.';
      });
    }
  }

  Future<void> playSong(Song song) async {
    final url = song.previewUrl;
    if (url == null || url.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Preview audio lagu ini tidak tersedia.'),
        ),
      );
      return;
    }

    try {
      if (playingSongId == song.id && audioPlayer.playing) {
        await audioPlayer.pause();
        if (mounted) setState(() {});
        return;
      }

      await audioPlayer.setUrl(url);
      if (!mounted) return;
      setState(() {
        currentSong = song;
        playingSongId = song.id;
      });
      unawaited(audioPlayer.play());
    } catch (e) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Audio gagal diputar. Silakan coba lagu lain.'),
        ),
      );
    }
  }

  void toggleFavorite(Song song) {
    setState(() {
      if (favoriteSongs.containsKey(song.id)) {
        favoriteSongs.remove(song.id);
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text('Lagu dihapus dari Library.'),
            duration: Duration(seconds: 1),
          ),
        );
      } else {
        favoriteSongs[song.id] = song;
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text('Lagu ditambahkan ke Library.'),
            duration: Duration(seconds: 1),
          ),
        );
      }
    });
  }

  void selectTab(int index) {
    FocusScope.of(context).unfocus();
    setState(() => currentIndex = index);
  }

  @override
  void dispose() {
    searchController.dispose();
    audioPlayer.dispose();
    super.dispose();
  }

  Widget artworkImage(Song song, {double size = 56}) {
    return ClipRRect(
      borderRadius: BorderRadius.circular(9),
      child: song.artwork.isEmpty
          ? Container(
              width: size,
              height: size,
              color: Colors.white12,
              child: const Icon(Icons.music_note),
            )
          : Image.network(
              song.artwork,
              width: size,
              height: size,
              fit: BoxFit.cover,
              errorBuilder: (_, __, ___) => Container(
                width: size,
                height: size,
                color: Colors.white12,
                child: const Icon(Icons.music_note),
              ),
            ),
    );
  }

  Widget songTile(Song song) {
    final isFavorite = favoriteSongs.containsKey(song.id);

    return Container(
      margin: const EdgeInsets.only(bottom: 9),
      decoration: BoxDecoration(
        color: cardColor,
        borderRadius: BorderRadius.circular(13),
      ),
      child: ListTile(
        contentPadding: const EdgeInsets.symmetric(horizontal: 10, vertical: 4),
        leading: artworkImage(song),
        title: Text(
          song.title,
          maxLines: 1,
          overflow: TextOverflow.ellipsis,
          style: const TextStyle(fontWeight: FontWeight.w600, fontSize: 14),
        ),
        subtitle: Text(
          song.artist,
          maxLines: 1,
          overflow: TextOverflow.ellipsis,
          style: const TextStyle(color: Colors.white60, fontSize: 12),
        ),
        trailing: SizedBox(
          width: 94,
          child: Row(
            mainAxisAlignment: MainAxisAlignment.end,
            children: [
              IconButton(
                tooltip: isFavorite ? 'Hapus dari Library' : 'Tambah ke Library',
                visualDensity: VisualDensity.compact,
                onPressed: () => toggleFavorite(song),
                icon: Icon(
                  isFavorite ? Icons.favorite : Icons.favorite_border,
                  color: isFavorite ? accentColor : Colors.white60,
                  size: 21,
                ),
              ),
              IconButton(
                tooltip: 'Putar preview',
                visualDensity: VisualDensity.compact,
                onPressed: () => playSong(song),
                icon: Icon(
                  playingSongId == song.id && audioPlayer.playing
                      ? Icons.pause_circle_filled
                      : Icons.play_circle_fill,
                  color: accentColor,
                  size: 31,
                ),
              ),
            ],
          ),
        ),
        onTap: () => playSong(song),
      ),
    );
  }

  Widget pageHeader(String title, String subtitle, IconData icon) {
    return Padding(
      padding: const EdgeInsets.fromLTRB(20, 22, 20, 18),
      child: Row(
        children: [
          Container(
            width: 46,
            height: 46,
            decoration: BoxDecoration(
              color: accentColor,
              borderRadius: BorderRadius.circular(14),
            ),
            child: Icon(icon, color: Colors.black, size: 27),
          ),
          const SizedBox(width: 12),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  title,
                  style: const TextStyle(
                    fontSize: 24,
                    fontWeight: FontWeight.bold,
                  ),
                ),
                const SizedBox(height: 3),
                Text(
                  subtitle,
                  style: const TextStyle(color: Colors.white60, fontSize: 12),
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }

  Widget loadingOrError({
    required bool loading,
    required String? error,
    required VoidCallback retry,
  }) {
    if (loading) {
      return const Padding(
        padding: EdgeInsets.all(42),
        child: Center(
          child: CircularProgressIndicator(color: accentColor),
        ),
      );
    }

    if (error != null) {
      return Padding(
        padding: const EdgeInsets.symmetric(horizontal: 28, vertical: 36),
        child: Center(
          child: Column(
            children: [
              const Icon(Icons.music_off, color: Colors.white54, size: 42),
              const SizedBox(height: 12),
              Text(error, textAlign: TextAlign.center),
              const SizedBox(height: 12),
              OutlinedButton(onPressed: retry, child: const Text('Coba lagi')),
            ],
          ),
        ),
      );
    }

    return const SizedBox.shrink();
  }

  Widget buildHomePage() {
    return RefreshIndicator(
      color: accentColor,
      onRefresh: fetchHomeSongs,
      child: ListView(
        physics: const AlwaysScrollableScrollPhysics(),
        padding: const EdgeInsets.fromLTRB(20, 10, 20, 20),
        children: [
          Row(
            children: [
              Container(
                width: 44,
                height: 44,
                decoration: BoxDecoration(
                  color: accentColor,
                  borderRadius: BorderRadius.circular(13),
                ),
                child: const Icon(
                  Icons.graphic_eq,
                  color: Colors.black,
                  size: 28,
                ),
              ),
              const SizedBox(width: 12),
              const Expanded(
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(
                      'SoundWave',
                      style: TextStyle(fontSize: 23, fontWeight: FontWeight.bold),
                    ),
                    Text(
                      'Music for every moment',
                      style: TextStyle(color: Colors.white60, fontSize: 12),
                    ),
                  ],
                ),
              ),
              const Icon(Icons.headphones, color: accentColor, size: 28),
            ],
          ),
          const SizedBox(height: 28),
          const Text(
            'Temukan musikmu',
            style: TextStyle(fontSize: 26, fontWeight: FontWeight.bold),
          ),
          const SizedBox(height: 7),
          const Text(
            'Temukan lagu baru untuk menemani harimu.',
            style: TextStyle(color: Colors.white60),
          ),
          const SizedBox(height: 22),
          InkWell(
            borderRadius: BorderRadius.circular(14),
            onTap: () => selectTab(1),
            child: Container(
              padding: const EdgeInsets.symmetric(horizontal: 15, vertical: 15),
              decoration: BoxDecoration(
                color: cardColor,
                borderRadius: BorderRadius.circular(14),
              ),
              child: const Row(
                children: [
                  Icon(Icons.search, color: Colors.white60),
                  SizedBox(width: 10),
                  Expanded(
                    child: Text(
                      'Cari judul lagu atau artis...',
                      style: TextStyle(color: Colors.white60),
                    ),
                  ),
                  Icon(Icons.arrow_forward_ios, size: 15, color: accentColor),
                ],
              ),
            ),
          ),
          const SizedBox(height: 28),
          const Text(
            'Lagu pilihan',
            style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
          ),
          const SizedBox(height: 13),
          if (isHomeLoading || homeError != null)
            loadingOrError(
              loading: isHomeLoading,
              error: homeError,
              retry: fetchHomeSongs,
            )
          else ...[
            Text(
              '${homeSongs.length} lagu dari iTunes',
              style: const TextStyle(color: Colors.white54, fontSize: 12),
            ),
            const SizedBox(height: 10),
            ...homeSongs.map(songTile),
          ],
        ],
      ),
    );
  }

  Widget buildSearchPage() {
    return ListView(
      padding: const EdgeInsets.fromLTRB(20, 10, 20, 20),
      children: [
        pageHeader('Search', 'Cari lagu dan artis favoritmu', Icons.search),
        TextField(
          controller: searchController,
          textInputAction: TextInputAction.search,
          onSubmitted: (_) => searchSongs(),
          decoration: InputDecoration(
            hintText: 'Judul lagu atau nama artis...',
            prefixIcon: const Icon(Icons.search),
            suffixIcon: IconButton(
              tooltip: 'Cari',
              onPressed: searchSongs,
              icon: const Icon(Icons.arrow_forward, color: accentColor),
            ),
            filled: true,
            fillColor: cardColor,
            border: OutlineInputBorder(
              borderRadius: BorderRadius.circular(14),
              borderSide: BorderSide.none,
            ),
          ),
        ),
        const SizedBox(height: 24),
        Text(
          lastSearchTerm.isEmpty ? 'Mulai pencarian' : 'Hasil untuk "$lastSearchTerm"',
          style: const TextStyle(fontSize: 19, fontWeight: FontWeight.bold),
        ),
        const SizedBox(height: 12),
        if (lastSearchTerm.isEmpty && !isSearchLoading)
          const Padding(
            padding: EdgeInsets.only(top: 28),
            child: Column(
              children: [
                Icon(Icons.travel_explore, size: 55, color: accentColor),
                SizedBox(height: 12),
                Text(
                  'Cari musik berdasarkan judul lagu atau nama artis.',
                  textAlign: TextAlign.center,
                  style: TextStyle(color: Colors.white60),
                ),
              ],
            ),
          )
        else if (isSearchLoading || searchError != null)
          loadingOrError(
            loading: isSearchLoading,
            error: searchError,
            retry: searchSongs,
          )
        else
          ...searchResults.map(songTile),
      ],
    );
  }

  Widget buildLibraryPage() {
    final favorites = favoriteSongs.values.toList();

    return ListView(
      padding: const EdgeInsets.fromLTRB(20, 10, 20, 20),
      children: [
        pageHeader('Library', 'Kumpulan lagu yang kamu sukai', Icons.library_music),
        if (favorites.isEmpty)
          Padding(
            padding: const EdgeInsets.only(top: 55),
            child: Column(
              children: [
                const Icon(Icons.favorite_border, size: 64, color: accentColor),
                const SizedBox(height: 16),
                const Text(
                  'Library kamu masih kosong',
                  style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
                ),
                const SizedBox(height: 8),
                const Text(
                  'Tekan ikon hati pada lagu untuk menyimpannya di sini.',
                  textAlign: TextAlign.center,
                  style: TextStyle(color: Colors.white60),
                ),
                const SizedBox(height: 20),
                FilledButton.icon(
                  onPressed: () => selectTab(0),
                  icon: const Icon(Icons.home_rounded),
                  label: const Text('Jelajahi lagu'),
                  style: FilledButton.styleFrom(
                    backgroundColor: accentColor,
                    foregroundColor: Colors.black,
                  ),
                ),
              ],
            ),
          )
        else ...[
          Text(
            '${favorites.length} lagu favorit',
            style: const TextStyle(color: Colors.white60, fontSize: 12),
          ),
          const SizedBox(height: 12),
          ...favorites.map(songTile),
        ],
      ],
    );
  }

  Widget buildMiniPlayer() {
    return StreamBuilder<bool>(
      stream: audioPlayer.playingStream,
      initialData: audioPlayer.playing,
      builder: (context, snapshot) {
        final isPlaying = snapshot.data ?? false;
        final song = currentSong;
        if (song == null) return const SizedBox.shrink();

        return Container(
          margin: const EdgeInsets.fromLTRB(10, 0, 10, 8),
          padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 8),
          decoration: BoxDecoration(
            color: const Color(0xFF303030),
            borderRadius: BorderRadius.circular(13),
            border: Border.all(color: Colors.white10),
          ),
          child: Row(
            children: [
              artworkImage(song, size: 42),
              const SizedBox(width: 10),
              Expanded(
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    Text(
                      song.title,
                      maxLines: 1,
                      overflow: TextOverflow.ellipsis,
                      style: const TextStyle(fontWeight: FontWeight.w600),
                    ),
                    Text(
                      song.artist,
                      maxLines: 1,
                      overflow: TextOverflow.ellipsis,
                      style: const TextStyle(color: Colors.white60, fontSize: 11),
                    ),
                    const Text(
                      'Preview iTunes Â· sekitar 30 detik',
                      style: TextStyle(color: accentColor, fontSize: 10),
                    ),
                  ],
                ),
              ),
              IconButton(
                tooltip: isPlaying ? 'Jeda' : 'Putar',
                onPressed: () async {
                  if (isPlaying) {
                    await audioPlayer.pause();
                  } else {
                    unawaited(audioPlayer.play());
                  }
                },
                icon: Icon(
                  isPlaying ? Icons.pause_rounded : Icons.play_arrow_rounded,
                  color: accentColor,
                  size: 30,
                ),
              ),
            ],
          ),
        );
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    final pages = [
      buildHomePage(),
      buildSearchPage(),
      buildLibraryPage(),
    ];

    return Scaffold(
      body: SafeArea(child: pages[currentIndex]),
      bottomNavigationBar: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          buildMiniPlayer(),
          NavigationBar(
            height: 66,
            backgroundColor: const Color(0xFF181818),
            indicatorColor: accentColor.withValues(alpha: 0.16),
            selectedIndex: currentIndex,
            onDestinationSelected: selectTab,
            destinations: const [
              NavigationDestination(
                icon: Icon(Icons.home_outlined),
                selectedIcon: Icon(Icons.home_rounded, color: accentColor),
                label: 'Home',
              ),
              NavigationDestination(
                icon: Icon(Icons.search_rounded),
                selectedIcon: Icon(Icons.search_rounded, color: accentColor),
                label: 'Search',
              ),
              NavigationDestination(
                icon: Icon(Icons.library_music_outlined),
                selectedIcon: Icon(Icons.library_music_rounded, color: accentColor),
                label: 'Library',
              ),
            ],
          ),
        ],
      ),
    );
  }
}
